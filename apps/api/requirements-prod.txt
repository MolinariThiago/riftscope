# RIFTSCOPE API - Production-only extras
#
# Install on top of requirements.txt when deploying to production
# (PostgreSQL + Redis + Celery + pandas). Versions here are kept
# CONSISTENT with the ranges in requirements.txt — install order
# doesn't matter, but the conventional flow is:
#
#   py -m pip install -r requirements.txt
#   py -m pip install -r requirements-prod.txt
#
# Anything that ALSO exists in requirements.txt (boto3, demoparser2,
# python-jose, bcrypt) is intentionally NOT pinned here a second time
# — duplicate pins are a recipe for resolver conflicts the day one
# side drifts. The version in requirements.txt is the single source
# of truth for those packages.

# PostgreSQL drivers — only needed when DATABASE_URL points at a
# real Postgres instance. SQLite dev installs don't need either.
asyncpg==0.30.0
psycopg2-binary==2.9.10

# Async task queue — required when QUEUE_BACKEND=celery. The inprocess
# backend in services/queue.py runs without these so they're a
# production-only ask.
celery==5.4.0
redis==5.2.1

# Data analysis — used by services/insights.py and services/maps.py
# for heavy aggregations. 2.2.x has wheels for Python 3.12 + 3.13.
pandas==2.2.3
